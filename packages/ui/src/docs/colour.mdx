# Colour

One primary, used at full strength. Large flat fields of `brand-primary` with white type on top —
that is the signature move. `brand-soft` is the calm counterpart; together with white it does 80% of
the work.

**Click any swatch, name or value to copy it** — the chip and the value copy the colour, the name
copies the token (`brand-primary`, `status-success`) for use in a class or prop.

**Rules.** Neutrals are warm, tinted toward the pink — never a blue-grey, never a pure `#888`.
Max one spice accent per screen beside the pink. Max two background colours per composition. Only
two gradients exist in the whole system, both legibility scrims over photography.

export const CopyIcon = ({ isCopied }) => (
  <svg
    aria-hidden="true"
    className="size-3.5 shrink-0 opacity-60"
    fill="none"
    stroke="currentColor"
    strokeLinecap="round"
    strokeLinejoin="round"
    strokeWidth="2"
    viewBox="0 0 24 24"
  >
    {isCopied ? (
      <polyline points="20 6 9 17 4 12" />
    ) : (
      <>
        <rect height="13" rx="2" ry="2" width="13" x="9" y="9" />
        <path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1" />
      </>
    )}
  </svg>
);

export const Swatch = ({ name, className, note }) => {
  const [value, setValue] = useState("");
  const [copied, setCopied] = useState("");

// Read the token's authored value straight off the custom property rather than hardcoding it
// here. A hardcoded hex is a second source of truth that silently drifts the moment a token
// changes — and it would put a literal brand hex outside `packages/design-tokens`, which the
// workspace forbids. This shows exactly what ships, including the rgba() tokens.
useEffect(() => {
const root = document.documentElement;
setValue(getComputedStyle(root).getPropertyValue(`--color-${name}`).trim());
}, [name]);

const copy = async (text, which) => {
await navigator.clipboard.writeText(text);
setCopied(which);
setTimeout(() => setCopied(""), 1200);
};

return (

<div className="sb-unstyled flex items-center gap-3">
<button
aria-label={`Copy the value ${value} for ${name}`}
className={`size-14 shrink-0 cursor-pointer rounded-3 border border-border-subtle ${className}`}
onClick={() => copy(value, "value")}
type="button"
/>
<div className="flex min-w-0 flex-col gap-0-5">
<button
aria-label={`Copy the token name ${name}`}
className="group flex cursor-pointer items-center gap-1-5 text-left font-mono text-mono tracking-mono text-text-heading hover:text-text-brand"
onClick={() => copy(name, "name")}
type="button" >
<span>{copied === "name" ? "copied" : name}</span>
<CopyIcon isCopied={copied === "name"} />
</button>
<button
aria-label={`Copy the value ${value}`}
className="flex cursor-pointer items-center gap-1-5 text-left font-mono text-caption tracking-mono text-text-muted hover:text-text-brand"
onClick={() => copy(value, "value")}
type="button" >
<span className="truncate">{copied === "value" ? "copied" : value || "\u2026"}</span>
<CopyIcon isCopied={copied === "value"} />
</button>
{note ? <span className="font-body text-caption text-text-subtle">{note}</span> : null}
</div>
</div>
);
};

export const Grid = ({ children }) => (
  <div className="grid grid-cols-[repeat(auto-fit,minmax(min(280px,100%),1fr))] gap-4 py-4">
    {children}
  </div>
);
